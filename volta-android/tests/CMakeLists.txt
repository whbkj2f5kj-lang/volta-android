cmake_minimum_required(VERSION 3.21)
project(VoltaPortRegressionTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network Test)
add_executable(mapproxy_test mapproxy_test.cpp)
target_link_libraries(mapproxy_test PRIVATE Qt6::Core Qt6::Gui Qt6::Network Qt6::Test)
enable_testing()
add_test(NAME mapproxy_test COMMAND mapproxy_test)
add_executable(places_test places_test.cpp)
target_link_libraries(places_test PRIVATE Qt6::Core Qt6::Gui Qt6::Network Qt6::Test)
add_test(NAME places_test COMMAND places_test)
option(VOLTA_GUI_TESTS "Test knob navigation with Qt Quick" OFF)
if(VOLTA_GUI_TESTS)
    find_package(Qt6 REQUIRED COMPONENTS Gui Qml Quick Location Positioning)
    add_executable(knob_test knob_test.cpp)
    target_link_libraries(knob_test PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Test)
    add_test(NAME knob_test COMMAND knob_test)
    set(_knob_env "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
    if(VOLTA_UPSTREAM_DIR)
        list(APPEND _knob_env "VOLTA_TEST_QML_DIR=${VOLTA_UPSTREAM_DIR}/qml")
    endif()
    set_tests_properties(knob_test PROPERTIES ENVIRONMENT "${_knob_env}")
    if(VOLTA_UPSTREAM_DIR)
        add_executable(raster_test raster_test.cpp)
        target_link_libraries(raster_test PRIVATE Qt6::Core Qt6::Gui Qt6::Network Qt6::Qml Qt6::Quick Qt6::Location Qt6::Positioning Qt6::Test)
        target_compile_definitions(raster_test PRIVATE VOLTA_PORT_TEST_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
        add_test(NAME raster_test COMMAND raster_test)
        set_tests_properties(raster_test PROPERTIES ENVIRONMENT "${_knob_env}" TIMEOUT 45)
    endif()
endif()
